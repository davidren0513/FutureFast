cmake_minimum_required(VERSION 3.16)
project(futurefast LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(ffhev SHARED IMPORTED)
set_target_properties(ffhev PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../libs/arm64-v8a/libffhev.so")
add_library(ffss STATIC IMPORTED)
set_target_properties(ffss PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/out/arm64-v8a/libfuturefast_ss.a")
add_library(futurefast SHARED native.cpp)
target_link_options(futurefast PRIVATE -Wl,--no-undefined)
target_link_libraries(futurefast PRIVATE ffhev ffss libace_napi.z.so pthread dl m)
